	Component* condition = comm->children;
	Component* then = condition->next;
	Component* elsse = then->next;
	//BOTH NUMBERS NO JUMPS NEEDED
	int help = compareA(condition->children, condition->children->next);
	if(help==1){
	buildCommands(then);
